import matplotlib
matplotlib.use('Agg')
import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt

# 1.1 Carga el dataset `titanic` de la librería seaborn.
df = sns.load_dataset('titanic')
print(df.head())

# 1.2 Forma del DataFrame, tipos de dato y resumen estadístico.
print(df.shape)
print(df.dtypes)
print(df.describe())

# 1.3 Valores nulos por columna, ordenados de mayor a menor.
nulos_por_columna = df.isnull().sum().sort_values(ascending=False)
print(nulos_por_columna)

# 1.4 Pasajeros por clase (pclass) y por puerto de embarque (embarked).
pasajeros_por_clase = df['pclass'].value_counts()
pasajeros_por_puerto = df['embarked'].value_counts()
print(pasajeros_por_clase)
print(pasajeros_por_puerto)

# 1.5 Combinaciones únicas class/embark_town con pd.crosstab.
tabla_unica = pd.crosstab(df['class'], df['embark_town'])
print(tabla_unica)
# No, no todas las clases embarcaron en todos los puertos (hay celdas en 0,
# p. ej. Second/Queenstown-Cherbourg son muy escasas). Se observa que en
# Cherbourg embarcó proporcionalmente más primera clase, mientras que en
# Southampton predomina claramente la tercera clase.

# 2.1 Media, mediana, desviación estándar y rango de fare (sin nulos).
fare_data = df['fare'].dropna().to_numpy()
media = np.mean(fare_data)
mediana = np.median(fare_data)
desviacion_std = np.std(fare_data)
rango = np.max(fare_data) - np.min(fare_data)
print(media, mediana, desviacion_std, rango)

# 2.2 Estandarización (z-score) de age, sin nulos. Primeros 5 valores.
age = df['age'].dropna().to_numpy()
z_score = (age - np.mean(age)) / np.std(age)
print(z_score[:5])

# 2.3 np.where: 'caro' si fare > media global, 'economico' en caso contrario.
fare_sin_nulos = df['fare'].dropna().to_numpy()
media_global = np.mean(fare_sin_nulos)
clasificacion = np.where(fare_sin_nulos > media_global, 'caro', 'economico')
valores, conteos = np.unique(clasificacion, return_counts=True)
print(dict(zip(valores, conteos)))

# 3.1 Copia df_clean sin filas con embarked nulo.
df_clean = df.copy()
df_clean = df_clean.dropna(subset=['embarked'])
print(df_clean.shape)

# 3.2 Rellena age con la mediana de su clase (pclass), no la mediana global.
df_clean['age'] = df_clean.groupby('pclass')['age'].transform(lambda x: x.fillna(x.median()))
print(df_clean['age'].isnull().sum())

# 3.3 Nueva columna family_size = sibsp + parch + 1.
df_clean['family_size'] = df_clean['sibsp'] + df_clean['parch'] + 1
print(df_clean['family_size'].head())

# 3.4 Pasajeros de primera clase con fare > 100 y sus puertos de embarque.
primera_clase_cara = df_clean[(df_clean['pclass'] == 1) & (df_clean['fare'] > 100)]
print(len(primera_clase_cara))
print(primera_clase_cara['embark_town'].value_counts())

# 4.1 Histograma de fare separando por clase (pclass) mediante color.
plt.figure(figsize=(8, 5))
sns.histplot(data=df_clean, x='fare', hue='pclass', multiple='stack')
plt.title('Distribución de fare por clase')
plt.show()

# 4.2 Pairplot de age, fare, family_size, sibsp coloreando por survived.
sns.pairplot(df_clean, vars=['age', 'fare', 'family_size', 'sibsp'], hue='survived')
plt.show()
# Interpretación: fare parece ser la variable que separa mejor visualmente a
# los supervivientes (tarifas más altas se asocian a mayor supervivencia);
# age y family_size aportan una separación mucho más débil.
